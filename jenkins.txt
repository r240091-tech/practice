    agent any

        pollSCM('H/5 * * * *')

        stage('Build') {
            steps {
                echo 'Building the project...'
                bat 'echo Build successful'

                echo 'Testing the project...'
                bat 'echo All tests passed'
            }
        }

            steps {
                echo 'Continuous Integration completed!'
            }
        }
    }

            echo 'Pipeline completed successfully.'
        }

        }
    }
